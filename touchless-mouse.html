1# Touchless Mouse Demo — MediaPipe + PyAutoGUI
2
3import cv2
4import mediapipe as mp
5import pyautogui
6import numpy as np
7
8# ── Config ──────────────────────────────────
9SCREEN_W, SCREEN_H = pyautogui.size()
10SMOOTHING = 0.2 # lower = smoother
11CLICK_THRESH = 0.04
12
13# ── MediaPipe Setup ──────────────────────────
14mp_hands = mp.solutions.hands
15hands = mp_hands.Hands(
16    max_num_hands=1,
17    min_detection_confidence=0.8,
18    min_tracking_confidence=0.7
19)
20
21def get_distance(p1, p2):
22    """Euclidean distance between two landmarks."""
23    return np.hypot(p1.x - p2.x, p1.y - p2.y)
24
25def fingers_up(lm):
26    """Return booleans for each finger being extended."""
27    tips = [8, 12, 16, 20] # index→pinky
28    return [lm[t].y < lm[t-2].y for t in tips]
29
30def main():
31    cap = cv2.VideoCapture(0)
32    px, py = 0, 0 # previous cursor pos
33    last_click = 0
34
35    while cap.isOpened():
36        ret, frame = cap.read()
37        if not ret: break
38        frame = cv2.flip(frame, 1)
39        rgb = cv2.cvtColor(frame, cv2.COLOR_BGR2RGB)
40        results = hands.process(rgb)
41
42        if results.multi_hand_landmarks:
43            lm = results.multi_hand_landmarks[0].landmark
44            up = fingers_up(lm)
45
46            # Move cursor: only index finger up
47            if up[0] and not up[1]:
48                nx = int(lm[8].x * SCREEN_W)
49                ny = int(lm[8].y * SCREEN_H)
50                cx = px + (nx-px)*SMOOTHING
51                cy = py + (ny-py)*SMOOTHING
52                pyautogui.moveTo(cx, cy)
53                px, py = cx, cy
54
55if __name__ == '__main__':
56    main()
